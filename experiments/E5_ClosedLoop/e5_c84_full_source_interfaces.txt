def eval_one(ds,gate_new,gate_frozen,split,variants=('original','matched_signal')):
 rows=[]
 for seed,(truth,variants_map,budget,alerts) in ds.items():
  for variant in variants:
   fr,nr=variants_map[variant]
   for name,rr,gg in [('Frozen',fr,gate_frozen['model']),('HardNegative',nr,gate_new)]:
    if not rr:continue
    pr=gg.predict_proba(np.asarray([r['features'] for r in rr]))[:,1]
    pred=top_budget(rr,pr,budget)
    tp=len(pred&truth)
    rows.append(dict(split=split,seed=seed,variant=variant,model=name,truth=len(truth),alerts=alerts,
                     candidate_children=len(rr),budget=budget,predicted=len(pred),tp=tp,fp=len(pred)-tp,fn=len(truth)-tp,
                     precision=tp/len(pred) if pred else 0.,recall=tp/len(truth),f1=f1(tp,len(pred),len(truth))))
 return rows

def rank_candidates(events,chosen,ranker):
    """Return child -> ordered parent IDs, using exactly the E5-C58 candidate rule."""
    selected=sorted((e for e in events if e['event_id'] in chosen),key=lambda e:(int(e['timestamp']),e['event_id']))
    active=deque(); output={}
    for child in selected:
        t=int(child['timestamp'])
        while active and t-int(active[0]['timestamp'])>W:active.popleft()
        pairs=[]
        for parent in reversed(active):
            gap=t-int(parent['timestamp'])
            if gap<=0:continue
            if not (parent['destination_host']==child['source_host'] or parent['source_host']==child['source_host'] or parent['destination_host']==child['destination_host']):continue
            pairs.append((parent,gap))
        pairs.sort(key=lambda z:(z[1],z[0]['event_id']))
        if pairs:
            x=[]
            for i,(parent,gap) in enumerate(pairs):
                x.append([gap/90.,float(parent['source_host']==child['source_host']),float(parent['destination_host']==child['destination_host']),float(parent['destination_host']==child['source_host']),float(parent['source_host']==child['destination_host']),float(parent['sensor']==child['sensor']),float(parent['event_type']==child['event_type']),float(parent['signal']),float(child['signal']),len(pairs)/50.,i/50.])
            x=np.asarray(x,dtype=float)
            z=x if len(ranker['features'])==11 else x[:,:-1]
            scores=np.asarray(ranker['model'].decision_function(z),dtype=float).reshape(-1)
            order=sorted(range(len(pairs)),key=lambda i:(-float(scores[i]),pairs[i][1],pairs[i][0]['event_id']))
            output[child['event_id']]=[pairs[i][0]['event_id'] for i in order]
        active.append(child)
    return output

def summarise(rows):

    for r in rows:grouped[tuple(r[k] for k in ('detector','variant','ranker','alerts','edge_budget'))].append(r)

          malicious_events=malicious,malicious_alerts=sum(r['malicious_alerts'] for r in group),
          event_recall=sum(r['malicious_alerts'] for r in group)/malicious if malicious else 0,
          alert_reachable_edges=alert_edges,alert_edge_recall=alert_edges/truth if truth else 0,
          candidate_reachable_edges=candidate,candidate_recall_given_alerts=candidate/alert_edges if alert_edges else 0,
          candidate_recall_global=candidate/truth if truth else 0,
          top1_correct=t1,top1_given_candidates=t1/candidate if candidate else 0,

def graph(events):
 """Host-compatible 1..90 time-neighborhood; symmetric for alert selection only."""
 n=len(events);order=sorted(range(n),key=lambda i:(int(events[i]['timestamp']),events[i]['event_id']))
 adj=[set() for _ in range(n)];near=[0]*n;active=deque()
 for i in order:
  a=events[i];t=int(a['timestamp'])
  while active and t-int(events[active[0]]['timestamp'])>W:active.popleft()
  for j in active:
   b=events[j];gap=t-int(b['timestamp'])
   if gap<=0:continue
   near[i]+=1;near[j]+=1
   if b['destination_host']==a['source_host'] or b['source_host']==a['source_host'] or b['destination_host']==a['destination_host']:
    adj[i].add(j);adj[j].add(i)
  active.append(i)
 return adj,near

def select(ids,probs,adj,near,k,alpha,beta):
 """Greedy score + bounded selected-neighbor bonus; deterministic, label-free.
 alpha rewards topology connections to already selected alerts; beta rewards temporal density.
 """
 n=len(ids);rank=sorted(range(n),key=lambda i:(-float(probs[i]),ids[i]))
 base=[0.]*n
 for pos,i in enumerate(rank):base[i]=1.-pos/max(n-1,1)
 if not alpha and not beta:return set(ids[i] for i in rank[:k])
 maxnear=max(max(near),1)
 time=[math.log1p(x)/math.log1p(maxnear) for x in near]
 count=[0]*n;selected=[False]*n;version=[0]*n
 def value(i):return base[i]+alpha*min(count[i],3)/3.+beta*time[i]
 heap=[(-value(i),ids[i],i,0) for i in range(n)];heapq.heapify(heap)

def mask(X,components,mapping):

def candidate_map(events,chosen):
 """Match E5-C58 parent compatibility: gap 1..90 and one of three host relations."""
 selected=sorted((r for r in events if r['event_id'] in chosen),key=lambda r:(int(r['timestamp']),r['event_id']))
 active=deque();candidates={}
 for child in selected:
  t=int(child['timestamp'])
  while active and t-int(active[0]['timestamp'])>W:active.popleft()
  cand=[]
  for parent in active:
   gap=t-int(parent['timestamp'])
   if gap<=0:continue
   a=int(parent['destination_host']==child['source_host'])
   b=int(parent['source_host']==child['source_host'])
   c=int(parent['destination_host']==child['destination_host'])
   if a or b or c:cand.append((parent['event_id'],gap,a,b,c))
  if cand:candidates[child['event_id']]=cand
  active.append(child)
 return candidates

def heuristics(cmap,probs,seed,variant):
 """Select one parent per child, then rank child edges by a label-free confidence score."""
 out={k:[] for k in METHODS[2:]}
 rng=random.Random(910000+seed+(0 if variant=='original' else 1))
 for child,candidates in cmap.items():
  # deterministic parent choice; ties broken by event ID
  recent=min(candidates,key=lambda x:(x[1],x[0]))
  handoff=min(candidates,key=lambda x:(-x[2],x[1],x[0]))
  combined=min(candidates,key=lambda x:(-(2*x[2]+x[3]+x[4]),x[1],x[0]))
  rand=candidates[rng.randrange(len(candidates))]
  for method,c in [('RecentParent',recent),('HostHandoff',handoff),('TemporalHost',combined),('SeededRandom',rand)]:
   parent,gap,a,b,d=c
   # Common child detector score ranks edge acceptance. Tiny deterministic tie-break
   # favors shorter gap and direct handoff; not tuned on any labels.
   confidence=float(probs[child])+1e-6*(W-gap)/W+1e-8*a
   out[method].append((confidence,parent,child))
 return out
