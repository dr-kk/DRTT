===== GENERATOR =====
import argparse, csv, json, random
from pathlib import Path

EVENT_TYPES = ['connection','login','process_start','file_access','configuration_change','service_request']
SENSORS = ['network','endpoint','authentication','application']
HOSTS = [f'host{i:02d}' for i in range(1,21)]


def write_csv(path, rows, fields):
    with path.open('w', newline='', encoding='utf-8') as f:
        w=csv.DictWriter(f, fieldnames=fields)
        w.writeheader(); w.writerows(rows)


def generate(seed, n_campaigns, duration, background_events):
    rng=random.Random(seed)
    raw=[]
    # The hidden campaign identifier and attack stage exist only in private truth.
    for c in range(n_campaigns):
        start=rng.randint(100, duration-400)
        hosts=rng.sample(HOSTS, 5)
        t=start
        parent=None
        for stage in range(1,8):
            t+=rng.randint(8,25)
            src,dst=([hosts[0],hosts[1]] if stage<=2 else
                     [hosts[1],hosts[2]] if stage<=4 else
                     [hosts[2],hosts[3]] if stage<=6 else
                     [hosts[3],hosts[4]])
            internal=f'attack_{c}_{stage}'
            raw.append(dict(internal=internal,t=t,src=src,dst=dst,
                            sensor=rng.choice(SENSORS),etype=rng.choice(EVENT_TYPES),
                            signal=round(max(0,min(1,rng.gauss(.68,.22))),5),
                            malicious=1,stage=stage,campaign=c,parent=parent))
            parent=internal
    for b in range(background_events):
        raw.append(dict(internal=f'benign_{b}',t=rng.randint(0,duration),
                        src=rng.choice(HOSTS),dst=rng.choice(HOSTS),
                        sensor=rng.choice(SENSORS),etype=rng.choice(EVENT_TYPES),
                        signal=round(max(0,min(1,rng.gauss(.47,.22))),5),
                        malicious=0,stage=0,campaign='',parent=None))
    # Sort chronologically, then replace internal IDs with opaque sequential IDs.
    # Note: monotone IDs are observable timestamps in disguise, not campaign identifiers.
    raw.sort(key=lambda r:(r['t'],rng.random()))
    idmap={r['internal']:f'EV{i:08d}' for i,r in enumerate(raw)}
    telemetry=[]; truth=[]
    for r in raw:
        eid=idmap[r['internal']]
        telemetry.append(dict(event_id=eid,timestamp=r['t'],source_host=r['src'],
                              destination_host=r['dst'],sensor=r['sensor'],
                              event_type=r['etype'],signal=r['signal']))
        truth.append(dict(event_id=eid,malicious=r['malicious'],stage=r['stage'],
                          campaign_id=r['campaign'],
                          predecessor_event_id=idmap[r['parent']] if r['parent'] else ''))
    return telemetry,truth


def audit(telemetry,truth,n_campaigns):
    assert len({r['event_id'] for r in telemetry})==len(telemetry)
    assert set(telemetry[0])=={'event_id','timestamp','source_host','destination_host','sensor','event_type','signal'}
    byid={r['event_id']:r for r in telemetry}
    groups={}
    for r in truth:
        if r['malicious']:
            groups.setdefault(r['campaign_id'],[]).append(r)
            if r['predecessor_event_id']:
                assert int(byid[r['predecessor_event_id']]['timestamp']) < int(byid[r['event_id']]['timestamp'])
    assert len(groups)==n_campaigns and all(len(g)==7 for g in groups.values())
    intervals=[(min(int(byid[r['event_id']]['timestamp']) for r in g),
                max(int(byid[r['event_id']]['timestamp']) for r in g)) for g in groups.values()]
    overlapping_pairs=sum(a[0]<=b[1] and b[0]<=a[1] for i,a in enumerate(intervals) for b in intervals[i+1:])
    timestamps=sorted(int(r['timestamp']) for r in telemetry)
    gaps=[b-a for a,b in zip(timestamps,timestamps[1:])]
    return {'events':len(telemetry),'malicious_events':sum(r['malicious'] for r in truth),
            'true_edges':sum(bool(r['predecessor_event_id']) for r in truth),
            'overlapping_campaign_pairs':overlapping_pairs,
            'maximum_timestamp_gap':max(gaps,default=0),
            'observable_columns':list(telemetry[0]),
            'note':'Synthetic correlated attacks; not executable causal response or independent real-world validation'}


def main():
    p=argparse.ArgumentParser()
    p.add_argument('--output',default='data/causal/e5c5')
    p.add_argument('--seed',type=int,default=2026)
    p.add_argument('--campaigns',type=int,default=100)
    p.add_argument('--duration',type=int,default=2400)
    p.add_argument('--background',type=int,default=10000)
    a=p.parse_args()
    if a.duration<=400 or a.campaigns<2 or a.background<1:
        p.error('duration must exceed 400, campaigns >= 2, background >= 1')
    out=Path(a.output);out.mkdir(parents=True,exist_ok=True)
    telemetry,truth=generate(a.seed,a.campaigns,a.duration,a.background)
    report=audit(telemetry,truth,a.campaigns)
    write_csv(out/'telemetry.csv',telemetry,list(telemetry[0]))
    write_csv(out/'ground_truth.csv',truth,list(truth[0]))
    (out/'audit.json').write_text(json.dumps(report,indent=2),encoding='utf-8')
    print('E5-C5 | OVERLAPPING CAMPAIGN GENERATION')
    print(json.dumps(report,indent=2))
    print('Saved:',out.resolve())
    if report['overlapping_campaign_pairs']==0:
        raise RuntimeError('No campaign overlap observed; change parameters')

if __name__=='__main__':main()

GATE_FEATURES=['best_score','score_margin','score_mean','score_std','log_candidate_count','best_gap_norm','best_parent_signal','child_signal','same_source','same_destination','dst_to_src','src_to_dst','same_sensor','same_type']

def read(p):
    with p.open(newline='',encoding='utf-8-sig') as f:return list(csv.DictReader(f))
def digest(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
    return h.hexdigest()
def extract(events,window):
    ordered=sorted(events,key=lambda r:(int(r['timestamp']),r['event_id']))
    history=deque();src=defaultdict(int);dst=defaultdict(int);pair=defaultdict(int);out={};i=0
    while i<len(ordered):
        t=int(ordered[i]['timestamp'])
        while history and history[0][0]<t-window:
            _,s,d=history.popleft();src[s]-=1;dst[d]-=1;pair[(s,d)]-=1
        j=i
        while j<len(ordered) and int(ordered[j]['timestamp'])==t:j+=1
        for r in ordered[i:j]:
            s,d=r['source_host'],r['destination_host']
            out[r['event_id']]=[float(r['signal']),src[s],dst[d],pair[(s,d)]]
        for r in ordered[i:j]:
            s,d=r['source_host'],r['destination_host'];history.append((t,s,d));src[s]+=1;dst[d]+=1;pair[(s,d)]+=1
        i=j
    return out

def top_predictions(events,chosen,ranker,gate,base,window):
    # Replicate E5-C15 candidates, replacing only selection; never replace raw signal.
    selected=sorted((r for r in events if r['event_id'] in chosen),key=lambda r:(int(r['timestamp']),r['event_id']))
    out=[]; candidate_count=0
    for j,child in enumerate(selected):
        ct=int(child['timestamp']);entries=[]
        for parent in reversed(selected[:j]):
            gap=ct-int(parent['timestamp'])
            if gap>window:break
            if gap<=0:continue
            if not (parent['destination_host']==child['source_host'] or parent['source_host']==child['source_host'] or parent['destination_host']==child['destination_host']):continue
            entries.append((parent,gap))
        entries.sort(key=lambda x:(x[1],x[0]['event_id']))
        if not entries:continue
        candidate_count+=1
        vectors=[]
        for ix,(p,gap) in enumerate(entries):
            vec=[gap/90.,float(p['source_host']==child['source_host']),float(p['destination_host']==child['destination_host']),float(p['destination_host']==child['source_host']),float(p['source_host']==child['destination_host']),float(p['sensor']==child['sensor']),float(p['event_type']==child['event_type']),float(p['signal']),float(child['signal']),len(entries)/50.,ix/50.]
            vectors.append(vec)
        scores=np.asarray(ranker['model'].decision_function(np.asarray(vectors)),dtype=float).reshape(-1)
        ranked=sorted(zip(entries,vectors,scores),key=lambda z:(-float(z[2]),z[0][1],z[0][0]['event_id']))
        ((p,gap),vec,top)=ranked[0]
        second=float(ranked[1][2]) if len(ranked)>1 else float(top)
        x=[float(top),float(top)-second,float(np.mean(scores)),float(np.std(scores)),float(np.log1p(len(entries))),float(gap)/float(window),float(vec[7]),float(vec[8]),*map(float,vec[1:7])]
        if len(x)!=len(GATE_FEATURES):raise RuntimeError('Gate feature dimension mismatch')
        out.append((p['event_id'],child['event_id'],x))
    if not out:return [],{'selected_events':len(selected),'candidate_children':candidate_count,'gate_probability_mean':None}
    probs=gate['model'].predict_proba(np.asarray([r[2] for r in out]))[:,1]
    kept=[(p,c) for (p,c,_),prob in zip(out,probs) if prob>=gate['cutoff']]
    return kept,{'selected_events':len(selected),'candidate_children':candidate_count,'gate_probability_mean':float(np.mean(probs))}

def metric(pred,truth,events):
    tp=len(set(pred)&truth);n=len(pred);p=tp/n if n else 0.;r=tp/len(truth) if truth else 0.
    return {'predicted':n,'tp':tp,'fp':n-tp,'precision':p,'recall':r,'f1':2*p*r/(p+r) if p+r else 0.,'false_edges_per_1000_events':(n-tp)*1000/len(events)}

def sha(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        for chunk in iter(lambda:f.read(1<<20),b''):h.update(chunk)
    return h.hexdigest()

def read(path):

def write_csv(path,rows,fields=None):

def top_rows(events,chosen,ranker,window):
    """Replicate E5-C38/E5-C56 context candidates, with rank index starting at zero."""
    selected=sorted((e for e in events if e['event_id'] in chosen),key=lambda e:(int(e['timestamp']),e['event_id']))
    rows=[]
    for j,child in enumerate(selected):
        t=int(child['timestamp']);pairs=[]
        for parent in reversed(selected[:j]):
            gap=t-int(parent['timestamp'])
            if gap>window:break
            if gap<=0:continue
            if not (parent['destination_host']==child['source_host'] or parent['source_host']==child['source_host'] or parent['destination_host']==child['destination_host']):continue
            pairs.append((parent,gap))
        pairs.sort(key=lambda z:(z[1],z[0]['event_id']))
        if not pairs:continue
        vec=[]
        for i,(parent,gap) in enumerate(pairs):
            vec.append([gap/90.,float(parent['source_host']==child['source_host']),float(parent['destination_host']==child['destination_host']),float(parent['destination_host']==child['source_host']),float(parent['source_host']==child['destination_host']),float(parent['sensor']==child['sensor']),float(parent['event_type']==child['event_type']),float(parent['signal']),float(child['signal']),len(pairs)/50.,i/50.])
        x=np.asarray(vec,dtype=float)
        if x.shape[1]!=11:raise ValueError('Candidate feature mismatch')
        z=x if len(ranker['features'])==11 else x[:,:-1]
        scores=np.asarray(ranker['model'].decision_function(z),dtype=float).reshape(-1)
        order=sorted(range(len(pairs)),key=lambda i:(-float(scores[i]),pairs[i][1],pairs[i][0]['event_id']))
        best=order[0];parent,gap=pairs[best];top=float(scores[best]);second=float(scores[order[1]]) if len(order)>1 else top
        v=x[best]
        features=[top,top-second,float(np.mean(scores)),float(np.std(scores)),float(np.log1p(len(pairs))),float(gap)/float(window),float(v[7]),float(v[8]),*map(float,v[1:7])]
        if len(features)!=len(GATE_FEATURES):raise ValueError('Gate feature mismatch')
        rows.append({'parent':parent['event_id'],'child':child['event_id'],'features':features})
    return rows

def truth_edges(path):
    truth=read(path)
    return {(r['predecessor_event_id'],r['event_id']) for r in truth if r['predecessor_event_id']}

def metrics(rows,truth,keep):
    selected=[r for r,k in zip(rows,keep) if k]
    tp=sum((r['parent'],r['child']) in truth for r in selected)
    n=len(selected);precision=tp/n if n else 0.;recall=tp/len(truth) if truth else 0.
    return {'predicted':n,'tp':tp,'fp':n-tp,'precision':precision,'recall':recall,'f1':2*precision*recall/(precision+recall) if precision+recall else 0.}

    for cutoff in sorted(set(map(float,probs))):
        keep=probs>=cutoff
        if int(keep.sum())<a.min_predictions:continue
        curve.append({'cutoff':cutoff,**metrics(new_rows,cal_truth,keep)})
    if not curve:raise RuntimeError('No admissible calibration thresholds')
    best=max(curve,key=lambda r:(r['f1'],r['precision'],r['cutoff']))
    baseline_m=metrics(old_rows,cal_truth,old_probs>=old_gate['cutoff'])
    new_m=metrics(new_rows,cal_truth,probs>=best['cutoff'])
    a.output.mkdir(parents=True,exist_ok=True)
    model_obj={'model':gate_model,'features':GATE_FEATURES,'cutoff':best['cutoff'],'e5c56_sha256':sha(a.retrained),'baseline_e5c15_sha256':sha(a.ranker),'threshold':.65,'window':90,'budget':a.budget,'calibration_dataset':'e5c17_calibration'}
    with (a.output/'compatible_gate.pkl').open('wb') as f:pickle.dump(model_obj,f)
    write_csv(a.output/'calibration_curve.csv',curve)
    audit={'experiment':'E5-C58','status':'TRAINING + DEVELOPMENT CALIBRATION ONLY; no independent evaluation','train_gate_rows':len(train_rows),'train_correct_top1':int(y.sum()),'train_incorrect_top1':int(len(y)-y.sum()),'calibration_alerts':alert_n,'calibration_children':len(new_rows),'calibration_true_edges':len(cal_truth),'budget_multiplier':a.budget,'baseline':baseline_m,'proposed':new_m,'selected_cutoff':best['cutoff'],'min_predictions':a.min_predictions,'training_selection':'signal >=0.65; gate training labels from e5c15_train only','calibration_selection':'context detector fitted on training split; top N from E5-C17 telemetry','sha256':{'baseline_ranker':sha(a.ranker),'new_ranker':sha(a.retrained),'old_gate':sha(a.gate),'train_telemetry':sha(a.train/'telemetry.csv'),'train_truth':sha(a.train/'ground_truth.csv'),'cal_telemetry':sha(a.calibration/'telemetry.csv'),'cal_truth':sha(a.calibration/'ground_truth.csv')},'limitations':['Calibration F1 is in-sample and optimistic','Gate training signal-selected but calibration context-selected: distribution shift possible','Frozen original gate calibration may shift at budget 1.5x','Original reranker and new reranker can select different parents','Same synthetic generator, no real-world validation','No E5-C57 or other independent test seeds used']}
    (a.output/'audit.json').write_text(json.dumps(audit,indent=2),encoding='utf-8')
    print('E5-C58 | TRAIN-ONLY COMPATIBLE GATE + DEVELOPMENT CALIBRATION')
    print(f'Train: {len(train_rows)} candidate children; correct top-1={int(y.sum())}; incorrect={int(len(y)-y.sum())}')
    print(f'Calibration: alerts={alert_n} children={len(new_rows)} true_edges={len(cal_truth)} budget=x{a.budget:g}')
    print('Baseline:',json.dumps(baseline_m))
    print('Proposed:',json.dumps(new_m))
    print(f'New gate cutoff: {best["cutoff"]:.8f}')
    print('Saved:',a.output.resolve())
    print('Do NOT claim independent improvement; reserve new seeds for E5-C59.')
